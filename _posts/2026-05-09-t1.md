​	使用 Global Mapper 进行数据栅格化或者转换成 GeoTiff 格式，并将空值设置为 0 值。

##### 一、点云导出为彩色栅格 gms 脚本

```
DIR_LOOP_START DIRECTORY="E:\TS3-5" FILENAME_MASKS="*.txt"
	IMPORT FILENAME=%FNAME_W_DIR%

IMPORT_ASCII FILENAME="%FNAME_W_DIR%" TYPE="POINT_ONLY" COORD_DELIM="AUTO" \
	 COORD_FORMAT="DECIMAL" COORD_ORDER="CUSTOM,1,2,3" INC_COORD_LINE_ATTRS="YES" COL_HEADERS="NO" \
	 INC_ELEV_COORDS="YES" CREATE_AREAS_FROM_LINES="NO" SKIP_COLUMNS="0" SKIP_ROWS="0" \
	 LINE_TYPE="Unclassified Line Feature" ELEV_UNITS="METERS" LABEL_FIELD_FORCE_OVERWRITE="NO" \
	 LABEL_FORMAT_NUMBERS="YES" LABEL_PRECISION="-1" LABEL_REMOVE_TRAILING_ZEROS="YES" \
	 LABEL_USE_SCIENTIFIC_NOTATION="NO" CLIP_COLLAR="NONE" AUTOSTYLE_COLOR="14070474,7323645,1841891"
GENERATE_ELEV_GRID FILENAME="%FNAME%" ELEV_UNITS="METERS" GRID_ALG="TIN" GRID_BIN_SIZE="3" \
	 LAYER_DESC="%FNAME% (Generated Grid)" NO_DATA_DIST_MULT="4" GRID_SAVE_TIN="NO" \
	 GRID_FILL_TO_BOUNDS="NO" GRID_FLATTEN_AREAS="YES" GRID_HEIGHTS_RELATIVE="NO" GRID_USE_CONSTRAINTS="YES" \
	 GRID_IGNORE_ZERO="NO" GRID_TYPE="ELEV"

EXPORT_RASTER FILENAME="E:\TS3-5\gray\%FNAME_WO_EXT%.tif" EXPORT_LAYER="%FNAME% (Generated Grid)" \
	 EXPORT_LAYER="%FNAME%" TYPE="COG" \
	 SAMPLING_METHOD="BICUBIC" NUM_BANDS="3" BAND_BIT_DEPTH="8" INC_VECTOR_DATA="NO" \
	 BG_TRANSPARENT="YES" COMPRESSION="LZW" ADD_OVERVIEW_LAYERS="NO" TILE_SIZE="0" TIFF_NO_GTIFF_HEADER="NO"
	 
	 UNLOAD_LAYER FILENAME="%FNAME%"
	 UNLOAD_LAYER FILENAME="%FNAME% (Generated Grid)"
	 
DIR_LOOP_END
```

> 如果需要点云填充更大的空隙，可以适当增大 `NO_DATA_DIST_MULT="4"` 变量的值

##### 三、栅格空值转换 bat 脚本

```
@echo off
chcp 65001 >nul
setlocal enabledelayedexpansion

:: ========== 配置区域 ==========
set INPUT_DIR=.
set OUTPUT_DIR=output
set SRC_NODATA=-32767
set DST_NODATA=0
set COMPRESSION=LZW
:: ==============================

:: 创建输出目录
if not exist "%OUTPUT_DIR%" (
    echo 创建输出目录: %OUTPUT_DIR%
    mkdir "%OUTPUT_DIR%"
)

:: 统计变量
set COUNT=0
set SUCCESS=0
set FAIL=0

echo ========================================
echo 开始处理 TIFF 文件
echo 输入目录: %INPUT_DIR%
echo 输出目录: %OUTPUT_DIR%
echo 源空值: %SRC_NODATA%
echo 目标空值: %DST_NODATA%
echo 压缩算法: %COMPRESSION%
echo ========================================
echo.

:: 循环处理
for %%f in ("%INPUT_DIR%\*.tif") do (
    set /a COUNT+=1
    set "filename=%%~nxf"
    echo [处理 !COUNT!] %%f
    
    gdalwarp ^
        -dstnodata %DST_NODATA% ^
        -srcnodata %SRC_NODATA% ^
				-co TILED=YES ^
				-co BLOCKXSIZE=256 ^
				-co BLOCKYSIZE=256 ^
        -co "COMPRESS=%COMPRESSION%" ^
				-co BIGTIFF=IF_NEEDED ^
        --config GTIFF_SRS_SOURCE EPSG ^
        "%%f" ^
        "%OUTPUT_DIR%\!filename!"
    
    if !errorlevel! equ 0 (
        set /a SUCCESS+=1
        echo [成功] !filename!
    ) else (
        set /a FAIL+=1
        echo [失败] !filename!
    )
    echo.
)

:: 输出统计信息
echo ========================================
echo 处理完成
echo Total: %COUNT%
echo Success: %SUCCESS%
echo Fail: %FAIL%
echo ========================================
pause
```

> 栅格如果要使用彩色带阴影的图像模式，空值部分需要是 0 值，不能是其它的值，否则无法透明。
